let musicPlaying = false function safeConfetti(){ if (typeof confetti !== "function") return confetti({ particleCount: 40, spread: 60, startVelocity: 20, origin: { y: 0.7 }, colors: ["#fff", "#ffe680", "#ffdf80", "#ffd966"] }) } function createPetal(){ const petal = document.createElement("div") petal.classList.add("petal") petal.innerText = ["🌼","💛","✨"][Math.floor(Math.random()*3)] petal.style.left = Math.random()*100 + "vw" petal.style.animationDuration = (Math.random()*3 + 3) + "s" document.body.appendChild(petal) setTimeout(() => petal.remove(), 7000) } function startMusic(){ const music = document.getElementById("bg-music") if(!music) return music.volume = 0.3 music.play().catch(()=>{}) } function toggleMusic(){ const music = document.getElementById("bg-music") const btn = document.getElementById("music-toggle") if(!music) return if(music.paused){ music.play() if(btn) btn.textContent = "🔊" }else{ music.pause() if(btn) btn.textContent = "🔇" } } window.addEventListener("load", () => { safeConfetti() setInterval(safeConfetti, 4000) setInterval(createPetal, 300) startMusic() }) // unlock audio document.addEventListener("click", startMusic, { once:true })
